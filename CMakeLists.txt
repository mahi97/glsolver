cmake_minimum_required(VERSION 3.20)
project(glcore LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(GLCORE_SANITIZE "Build with address/undefined sanitizers" OFF)
option(GLCORE_DEBUG_ASSERTS "Enable expensive invariant assertions in the C++ core" OFF)

find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)

file(GLOB GLCORE_SOURCES CONFIGURE_DEPENDS src/glcore/*.cpp)
pybind11_add_module(_core MODULE ${GLCORE_SOURCES})
target_include_directories(_core PRIVATE src/glcore)
target_compile_options(_core PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  target_compile_options(_core PRIVATE -O3)
endif()
if(GLCORE_SANITIZE)
  target_compile_options(_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g -O1)
  target_link_options(_core PRIVATE -fsanitize=address,undefined)
endif()
if(GLCORE_DEBUG_ASSERTS)
  target_compile_definitions(_core PRIVATE GLCORE_DEBUG_ASSERTS=1)
endif()
install(TARGETS _core LIBRARY DESTINATION glsolver)
